Super clean zero-runtime deep immutable patch utility in TypeScript 5.4. Preserves discriminated unions without requiring Immer or Lodash.

immutable-patch.ts 8 lines • 248 bytes 
export type DeepPatch = {
  [K in keyof T]?: T[K] extends object ? DeepPatch : T[K];
};

export const immutablePatch = extends object>(base: T, patch: DeepPatch): T => {
  return Object.entries(patch).reduce((acc, [key, val]) => {
    acc[key as keyof T] = typeof val === 'object' && val !== null
      ? immutablePatch(acc[key as keyof T] as any, val)
      : val;
    return acc;
  }, { ...base });
};
